\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm]
\item You flip a coin and roll a standard $6$-sided die.
\begin{itemize}
\item What is the probability of getting Heads? \blank
\item What is the probability of rolling a $4$? \blank
\end{itemize}
\end{enumerate}

\newpage

% --- PART B ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part B: Calculating Compound Probability directly}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]
\begin{itemize}
\item Multiply them together: What is $P(\text{Heads AND a 4})$? \blank
\end{itemize}
\item A jar has $3$ red marbles and $2$ blue marbles. You pick one, PUT IT BACK, and pick again.
\begin{itemize}
\item What is $P(\text{Red})$ on the first pick? \blank
\end{itemize}
\end{enumerate}

\newpage

% --- PART C ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part C: Calculating Compound Probability directly}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]
\begin{itemize}
\item What is $P(\text{Blue})$ on the second pick? \blank
\item What is $P(\text{Red, then Blue})$? \blank
\end{itemize}
\item Using the same jar as above ($3$ red, $2$ blue). You pick one marble, KEEP IT, and pick again.
\end{enumerate}

\newpage

% --- PART D ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part D: Calculating Compound Probability directly}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]
\item Challenge: What is $P(\text{Red})$ on the first pick? \blank
\item If you kept that red marble, how many total marbles are left? \blank
\item Now what is the new probability of picking a Blue marble on the second pick? \blank
\item Multiply to find $P(\text{Red, then Blue WITHOUT replacement})$. \blank
\end{enumerate}
