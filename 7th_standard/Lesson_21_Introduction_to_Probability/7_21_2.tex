% PART A - UNDERSTANDING 'NOT'
% ==========================================================
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part A: The Probability of an Event NOT Happening}}
\end{tcolorbox}

\vspace{3mm}

\noindent
The probability of an event happening PLUS the probability of it NOT happening always equals $1$ (or $100\%$). 
\[ P(\text{Event}) + P(\text{Not Event}) = 1 \]

\vspace{0.5cm}

\begin{enumerate}[itemsep=\fill, leftmargin=0.6cm]

\item The probability of rain tomorrow is $40\%$. What is the probability that it will NOT rain? \blank

\item You have a deck of $52$ cards. The probability of drawing a heart is $\frac{1}{4}$. What is the probability of NOT drawing a heart? \blank

\item A weather forecaster says there is a $0.15$ chance of snow. What is the chance of no snow? \blank

\end{enumerate}

% ==========================================================
% PART B - CRITICAL THINKING WITH DATA

\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part B: Analyzing Experimental Data}}
\end{tcolorbox}

\noindent
Read the scenario and analyze the probabilities.

\vspace{0.5cm}

\begin{enumerate}[itemsep=\fill, leftmargin=0.6cm, resume]

\item A student flips a coin $50$ times and records the results. They get Heads $35$ times and Tails $15$ times.
\begin{itemize}
    \item What is the experimental probability of getting Heads? \blank
    \item What is the theoretical probability of getting Heads? \blank
    \item Do these two probabilities match? If the student flipped the coin $10,000$ times instead, would you expect the experimental probability of Heads to stay exactly at $\frac{35}{50}$ ($70\%$)? Why or why not?
\end{itemize}

\end{enumerate}
