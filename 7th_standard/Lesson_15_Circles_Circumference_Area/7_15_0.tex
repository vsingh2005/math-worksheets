\end{tcolorbox}

\vspace{3mm}

\noindent
\textbf{Key Terms:}
\begin{itemize}
    \item \textbf{Radius ($r$):} The distance from the center to the edge.
    \item \textbf{Diameter ($d$):} The distance straight across the circle, through the center. ($d = 2r$)
    \item \textbf{Circumference ($C$):} The distance around the outside of the circle (the perimeter).
    \item \textbf{Area ($A$):} The amount of space inside the circle.
\end{itemize}

\vspace{2mm}
\noindent
\textbf{Formulas (Use $\pi \approx 3.14$):}
\begin{itemize}
    \item \textbf{Circumference:} $C = \pi d$ \quad or \quad $C = 2 \pi r$
    \item \textbf{Area:} $A = \pi r^2$
\end{itemize}
