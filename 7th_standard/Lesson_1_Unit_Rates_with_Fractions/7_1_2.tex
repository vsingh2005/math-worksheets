% --- PART A ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part A: Find the Unit Rate}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=4.2cm, leftmargin=0.6cm]
\item A snail moves $\dfrac{1}{5}$ of a meter in $\dfrac{1}{3}$ of a minute. What is its speed in meters per minute?
\item A recipe calls for $\dfrac{2}{5}$ cup of milk to make $\dfrac{1}{4}$ of a pitcher of pancakes. How many cups of milk are needed for $1$ full pitcher?
\item A cleaning crew cleans $\dfrac{5}{6}$ of an office building in $\dfrac{1}{2}$ of a day. How much of the building do they clean per day?
\end{enumerate}

\newpage

% --- PART B ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part B: Find the Unit Rate}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=4.2cm, leftmargin=0.6cm, resume]
\item A runner jogs $\dfrac{3}{8}$ of a mile in $\dfrac{1}{5}$ of an hour. What is their speed in miles per hour?
\item A 3D printer prints $\dfrac{7}{10}$ of a model in $\dfrac{1}{4}$ of an hour. How many models does it print per hour?
\item \textbf{Brand A} detergent costs $\$4$ for $\dfrac{1}{3}$ of a gallon. \\
\textbf{Brand B} detergent costs $\$5$ for $\dfrac{1}{2}$ of a gallon. \\ \\
Which brand is the better buy? (Find the cost per gallon)
\end{enumerate}

\newpage

% --- PART C ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part C: Find the Unit Rate}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=4.2cm, leftmargin=0.6cm, resume]
\item \textbf{Runner X} completes $\dfrac{2}{3}$ of a lap in $\dfrac{1}{4}$ of a minute. \\
\textbf{Runner Y} completes $\dfrac{3}{5}$ of a lap in $\dfrac{1}{6}$ of a minute. \\ \\
Which runner is faster?
\item \textbf{Painter 1} paints $\dfrac{1}{4}$ of a room in $\dfrac{1}{3}$ of an hour. \\
\textbf{Painter 2} paints $\dfrac{2}{5}$ of a room in $\dfrac{1}{2}$ of an hour. \\ \\
Who paints more of the room per hour?
\item An athlete drinks $\dfrac{3}{4}$ of a liter of water every $\dfrac{1}{2}$ hour of exercise.
\end{enumerate}

\newpage

% --- PART D ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part D: Find the Unit Rate}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=4.2cm, leftmargin=0.6cm, resume]
\item How much water does the athlete drink per hour?
\item If the athlete exercises for $3$ hours, how much water will they drink?
\item A factory machine produces $\dfrac{4}{7}$ of a batch of toys in $\dfrac{1}{5}$ of an hour.
\begin{itemize}
\item How many batches of toys are produced per hour?
\item How long will it take to produce $4$ full batches?
\end{itemize}
\end{enumerate}
