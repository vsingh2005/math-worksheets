% ==========================================================
% PART C — REAL-WORLD APPLICATIONS
% ==========================================================
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part C: Real-World Applications}}
\end{tcolorbox}

\vspace{3mm}

\noindent
Read each problem carefully. Write an expression and solve it.

\vspace{3mm}

\begin{enumerate}[itemsep=\fill, leftmargin=0.6cm]

\item The temperature was $-5^\circ C$ in the morning. By noon, it rose $12^\circ C$. By evening, it dropped $7^\circ C$. What was the final temperature?
\vspace{1.5cm}

\item A submarine is at an elevation of $-250.5$ meters. It ascends $120.75$ meters, then descends $80.25$ meters. What is its final elevation?
\vspace{1.5cm}

\item Sarah has a bank balance of $\$45.75$. She writes a check for $\$60.50$ and then deposits $\$35.25$. What is her new bank balance?
\vspace{1.5cm}

\item A person owes their friend $\$15.50$. They pay back $\$7.25$ but then borrow another $\$10.00$. What is their current financial standing with their friend? (Use a rational number to represent this).
\vspace{1.5cm}

\item A hiker starts at an elevation of $\dfrac{1}{2}$ mile above sea level. They descend $\dfrac{3}{4}$ mile into a canyon, then climb up $\dfrac{1}{3}$ mile. What is their final elevation relative to sea level?
\vspace{1.5cm}

\end{enumerate}
