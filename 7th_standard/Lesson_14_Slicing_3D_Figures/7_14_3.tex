\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm]

\end{enumerate}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]

\item The parallel cross section of a pyramid gets smaller as the slice is made closer to the top.
\end{enumerate}
