\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Grid Distance Challenge}}
\end{tcolorbox}

\noindent
Solve each multi-step coordinate plane problem.

\vspace{3mm}
\begin{enumerate}[itemsep=1.5cm, leftmargin=0.6cm]
\item A map is drawn on a coordinate plane where 1 unit represents 10 miles. \\
\textbf{a)} Town A is at $(-3, 4)$ and Town B is at $(5, 4)$. What is the distance between Town A and Town B in miles? \blanklong \\
\textbf{b)} Town C is at $(5, -2)$. What is the distance from Town B to Town C in miles? \blanklong \\
\textbf{c)} A helicopter flies directly from Town A to Town B, and then from Town B to Town C. What is the total distance traveled? \blanklong
\vspace{1.5cm}
\item Explain why points $P(-3, 2)$ and $Q(3, 2)$ are reflections of each other across the $y$-axis. Describe their distance on the number line in relation to 0.
\vspace{1.5cm}
\end{enumerate}
