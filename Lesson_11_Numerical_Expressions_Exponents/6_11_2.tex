% ==========================================================
% PART C — COMPARING AND MATCHING
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Comparing and Matching}}
\end{tcolorbox}

\noindent
Find and compare the values of the following expressions.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item Compare: $2 \times 3^2 + 5$ \quad and \quad $(2 \times 3)^2 + 5$. Which has a larger value, and why?

\vspace{2cm}

\item Compare: $2 \times (3^2 + 5)$ \quad and \quad $2^2 \times 3 + 5$. Show all calculations.

\vspace{2cm}

\item Determine which expression has the greatest value: $4 \times 5 - 2$, \ $4 \times (5 - 2)$, \ or \ $4 \times (5 - 2^2)$.

\vspace{2cm}

\end{enumerate}

\large\textcolor{black}{\textbf{Intermediate Word Problems}}

\noindent
Write a numerical expression for each situation and solve it.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item A group of 4 friends buys movie tickets for \$10 each and then buys a large popcorn to share for \$8. Write a numerical expression for the total cost, and evaluate it.

\vspace{2.2cm}

\item A storage box contains 3 trays of chocolates. Each tray holds a grid of $4 \times 4$ chocolates. Write an expression using exponents for the total number of chocolates, and evaluate it.

\vspace{2.2cm}

\item A classroom has 5 rows of 6 desks. Before the class starts, 3 desks are removed. Write an expression for the remaining desks, and evaluate it.

\vspace{2.2cm}

\end{enumerate}
