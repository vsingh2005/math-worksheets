% ==========================================================
% PART C — WRITING EQUATIONS FROM WORD PROBLEMS
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Writing Equations from Word Problems}}
\end{tcolorbox}

\noindent
Write a two-variable equation for each scenario.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item An online bookstore charges \$5 shipping plus \$4 per book $b$ ordered. Write an equation for the total cost $C$.

\vspace{2cm}

\item A plant's height $h$ in cm after $w$ weeks is modeled by a starting height of 10 cm and growing 2 cm per week. Write an equation.

\vspace{2cm}

\item A pool contains 500 gallons of water and drains at a rate of 25 gallons per hour $t$. Write an equation for the remaining water $W$.

\vspace{2cm}

\end{enumerate}

\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Real-World Graphing Reasoning}}
\end{tcolorbox}

\noindent
Write inequalities and describe their number line graphs.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item ``The temperature in the freezer must be below $0^\circ\text{C}$.'' Write an inequality using $t$ and describe its graph.

\vspace{2.2cm}

\item ``A cargo van can carry a maximum load of 1,500 pounds.'' Write an inequality using $w$ and describe its graph.

\vspace{2.2cm}

\item ``A credit card requires a minimum payment of \$25.'' Write an inequality using $p$ and describe its graph.

\vspace{2.2cm}

\end{enumerate}
