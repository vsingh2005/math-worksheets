% ==========================================================
\begin{tcolorbox}[colback=black!10]
    \large\textcolor{black}{\textbf{Area of Irregular and Composite Polygons (Foundation)}}
\end{tcolorbox}

\vspace{3mm}

\noindent
1. \textbf{Decomposition:} Break down complex or irregular polygons into simpler shapes (rectangles, triangles, parallelograms) whose areas are easy to calculate. Then, sum the areas of the parts. \\
2. \textbf{Subtraction Method:} Draw a bounding rectangle around the shape, find its total area, and subtract the areas of the empty corners. \\
3. \textbf{Coordinate Grid Polygons:} Find area by calculating the vertical and horizontal lengths of the decomposed pieces on the coordinate plane.

\vspace{4mm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=5mm, bottom=5mm]
\textbf{Worked Example:}

\medskip
\noindent
An L-shaped polygon has a height of 10 cm, a total width of 8 cm. The vertical column is 4 cm wide, and the horizontal base is 3 cm high. Find its area.\\
\textbf{Solution:} \\
1) Decompose into two rectangles:\\
- Vertical rectangle: $Width = 4\text{ cm}$, $Height = 10 - 3 = 7\text{ cm}$. $\text{Area}_1 = 4 \times 7 = 28\text{ cm}^2$.\\
- Horizontal base: $Width = 8\text{ cm}$, $Height = 3\text{ cm}$. $\text{Area}_2 = 8 \times 3 = 24\text{ cm}^2$.\\
2) Add the areas: $\text{Total Area} = 28 + 24 = 52\text{ cm}^2$.
\end{tcolorbox}

\vspace{4mm}
