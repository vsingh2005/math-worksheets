\begin{tcolorbox}[colback=black!10]
    \large\textcolor{black}{\textbf{Area of Triangles and Parallelograms (Foundation)}}
\end{tcolorbox}

\vspace{3mm}

\noindent
1. \textbf{Parallelogram:} $\text{Area} = \text{base} \times \text{height} = b \times h$. \\
2. \textbf{Triangle:} $\text{Area} = \frac{1}{2} \times \text{base} \times \text{height} = \frac{1}{2}bh$. \\
- \textit{Concept:} A triangle is exactly half of a parallelogram with the same base and height.

\vspace{4mm}
\vspace{1cm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=5mm, bottom=5mm]
\textbf{Worked Example 1:}

\medskip
\noindent
Find the area of a triangle with a base of 6 cm and a height of 4 cm.\\
\textbf{Solution:} \\
$$\text{Area} = \frac{1}{2} \times 6 \times 4 = 12\text{ cm}^2.$$
\end{tcolorbox}

\vspace{4mm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=5mm, bottom=5mm]
\textbf{Worked Example 1B: Algebraic Dimension Matching}

\medskip
\noindent
\textbf{Problem:} A triangular piece of sheet metal has an area of 36 cm$^2$. Its height is exactly twice its base length. Find both the base and height dimensions.

\medskip
\noindent
\textbf{Solution:} \\
1) Assign variables based on the rule: Let base $= b$, then height $h = 2b$. \\
2) Substitute these definitions directly into your area formula:
$$\text{Area} = \frac{1}{2} \cdot b \cdot h \implies 36 = \frac{1}{2} \cdot b \cdot (2b)$$
3) Simplify the algebraic expression:
$$36 = 1b^2 \implies b = \sqrt{36} = 6\text{ cm}$$
4) Calculate the remaining dimension: $h = 2(6) = 12\text{ cm}$.
\end{tcolorbox}
\vspace{4mm}

% ==========================================================
\begin{tcolorbox}[colback=black!10]
    \large\textcolor{black}{\textbf{Area of Trapezoids and Decomposing Polygons (Foundation)}}
\end{tcolorbox}

\vspace{3mm}

\noindent
1. \textbf{Trapezoid:} $\text{Area} = \frac{1}{2}(b_1 + b_2)h$. \\
- \textit{Concept:} The area is the average of the two parallel bases multiplied by the height. \\
2. \textbf{Decomposing Polygons:} Any complex polygon can be split (decomposed) into triangles and rectangles to calculate its total area.

\vspace{4mm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=5mm, bottom=5mm]
\textbf{Worked Example 2:}

\medskip
\noindent
Find the area of a trapezoid with parallel bases of 8 cm and 12 cm, and a height of 5 cm.\\
\textbf{Solution:} \\
$$\text{Area} = \frac{1}{2}(8 + 12) \times 5 = \frac{1}{2}(20) \times 5 = 10 \times 5 = 50\text{ cm}^2.$$
\end{tcolorbox}

\vspace{4mm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=5mm, bottom=5mm]
\textbf{Worked Example 2B: Decomposition via Subtraction}

\medskip
\noindent
\textbf{Problem:} A large wooden rectangular billboard measuring 12 feet by 8 feet has a triangular logo cut out from its center. The triangular cutout has a base of 4 feet and a height of 3 feet. Find the remaining surface area of the solid billboard.

\medskip
\noindent
\textbf{Solution:} \\
1) Calculate the full outer bounding area (Rectangle):
$$\text{Area}_{\text{rectangle}} = \text{base} \times \text{height} = 12 \times 8 = 96\text{ ft}^2$$
2) Calculate the inner cutout area (Triangle):
$$\text{Area}_{\text{triangle}} = \frac{1}{2} \times \text{base} \times \text{height} = \frac{1}{2} \times 4 \times 3 = 6\text{ ft}^2$$
3) Subtract the cutout from the whole piece to find the remaining surface area:
$$\text{Area}_{\text{total}} = 96 - 6 = 90\text{ ft}^2$$
\end{tcolorbox}
\vspace{4mm}
