\end{enumerate}

% ==========================================================

% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Basic Word Problems}}
\end{tcolorbox}

\noindent
Write a one-step equation for each scenario and solve it.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item A boy has 5 marbles. He buys some more marbles, $m$, and now has 12. Write an equation and solve for $m$.

\vspace{2.2cm}

\item You divide a pack of candies evenly among 4 friends. Each friend gets 6 candies. Write a division equation using $c$ (total candies) and solve it.

\vspace{2.2cm}

\item A tree grows 3 inches in one year. Its new height is 45 inches. Write an equation using $h$ (starting height) and solve it.

\vspace{2.2cm}

\end{enumerate}
