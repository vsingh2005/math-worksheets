% ==========================================================
% PART C — SOLUTION OF INEQUALITIES
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Solution of Inequalities}}
\end{tcolorbox}

\noindent
Verify if the given value of the variable makes the inequality statement true.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item Is $x = 5$ a solution to $x > 5$? Explain.

\vspace{2cm}

\item Is $y = -2$ a solution to $y \le 0$? Explain.

\vspace{2cm}

\item Is $z = 10$ a solution to $z \ge 8$? Explain.

\vspace{2cm}

\item Is $j = 202.5$ a solution to $j \le 203$? Explain.

\vspace{2cm}

\end{enumerate}

\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Basic Word Problems}}
\end{tcolorbox}

\noindent
Write inequalities to model each real-world boundary.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item A child must be at least 42 inches tall ($h$) to go on a water slide. Write an inequality.

\vspace{2.2cm}

\item A box can hold at most 25 pounds of books ($w$). Write an inequality.

\vspace{2.2cm}

\item An elevator has a limit of 15 people ($p$). Write an inequality.

\vspace{2.2cm}

\item A crayon box can only hold 35 crayons. ($c$). Write an inequality.

\vspace{2.2cm}

\end{enumerate}
