\noindent
Write the reflected coordinates of the polygon vertices.

\vspace{3mm}
\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]
\item A rectangle has vertices $J(1, 2)$, $K(5, 2)$, $L(5, -1)$, and $M(1, -1)$. \\
\textbf{a)} If reflected across the $y$-axis, write the new vertices $J'$, $K'$, $L'$, and $M'$. \vspace{2cm} \\
\textbf{b)} Find the perimeter of the rectangle. 

\vspace{0.4cm}

\item Triangle $PQR$ has vertices $P(-2, 1)$, $Q(3, 1)$, and $R(3, 4)$. \\
\textbf{a)} Reflect it across the $x$-axis. Write the new vertices $P'$, $Q'$, $R'$.  \\ \vspace{2cm} \\
\textbf{b)} What type of triangle is $PQR$? 

\vspace{2cm}

\end{enumerate}

% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Reasoning with Reflections}}
\end{tcolorbox}

\noindent
Explain and solve.

\vspace{3mm}
\begin{enumerate}[itemsep=2.3cm, leftmargin=0.6cm]
\item Explain why points $P(-3, 2)$ and $Q(3, 2)$ are reflections across the $y$-axis. What is the distance between them?

\vspace{2.2cm}

\item A point is reflected across the $x$-axis and then across the $y$-axis. Is the final position the same as reflecting across both axes simultaneously? Explain with an example.

\vspace{2.2cm}

\item A point $A(5, -3)$ is reflected across the $x$-axis to get $A'$. Then $A'$ is reflected across the $y$-axis to get $A''$. In which quadrant is $A''$?

\vspace{2.2cm}

\end{enumerate}
