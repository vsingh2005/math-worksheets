% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Magnitude and Ordering Reasoning}}
\end{tcolorbox}

\noindent
Solve each problem.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item You are checking freezer temperatures in a scientific lab. Freezer A is at $-25^\circ\text{C}$, Freezer B is at $-18^\circ\text{C}$, and Freezer C is at $-30^\circ\text{C}$. \\
\textbf{a)} List the freezers in order from warmest to coldest. \blanklong \\
\textbf{b)} Which freezer requires the most cooling power? Explain using absolute value.

\vspace{2cm}

\item An accountant writes down two ledger entries: $-\$450$ (withdrawal) and $-\$120$ (service fee). \\
\textbf{a)} Write an inequality to show which entry represents a lower balance. \blanklong \\
\textbf{b)} Use absolute value to express which entry represents a larger reduction in funds.

\vspace{2cm}

\item A meteorologist notes that the low temperature on Monday was $-8^\circ\text{F}$ and on Tuesday was $-15^\circ\text{F}$. Write an inequality comparing these temperatures, and use absolute values to discuss which day had the colder temperature.

\vspace{2cm}

\end{enumerate}
