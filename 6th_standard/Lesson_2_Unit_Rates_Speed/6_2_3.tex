\begin{enumerate}[itemsep=2cm, leftmargin=0.6cm]

\item How many notebooks can you buy with \$25? \blank \; notebooks

\begin{itemize}
    \item Small: 20 oz for \$3
    \item Medium: 32 oz for \$4.50
    \item Large: 48 oz for \$6
\end{itemize}
Which size gives the best price per ounce? Show all calculations.
\vspace{3.5cm}

\end{enumerate}
