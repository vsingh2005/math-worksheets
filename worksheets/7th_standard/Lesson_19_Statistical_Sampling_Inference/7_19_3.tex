% --- PART C ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part C: Estimating Wildlife Populations}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]
\item Set up a proportion: $\dfrac{\text{Tagged Animals in 2nd Sample}}{\text{Total Animals in 2nd Sample}} = \dfrac{\text{Total Tagged Initially}}{\text{Estimated Total Population}}$
\end{enumerate}

\newpage

% --- PART D ---
\begin{tcolorbox}[colback=customteal!20]
\large\textcolor{black}{\textbf{Part D: Estimating Wildlife Populations}}
\end{tcolorbox}

\vspace{3mm}

\begin{enumerate}[itemsep=5.5cm, leftmargin=0.6cm, resume]
\item Researchers catch and tag $60$ fish in a lake. A week later, they catch a second sample of $80$ fish. Of those $80$ fish, $12$ have tags. Estimate the total number of fish in the lake.
\item Park rangers tag $45$ deer in a forest. Later, they observe $100$ deer and notice that $15$ of them are tagged. Estimate the total deer population.
\item A streaming service wants to know what genre of movies its users prefer. They email a survey to all $100,000$ users. Only $500$ people reply, and $80\%$ of them say "Horror". Can the company safely conclude that $80\%$ of ALL their users prefer Horror movies? What is the problem with this type of sample? (Hint: Think about who actually takes the time to reply).
\end{enumerate}
