% ==========================================================
% PART B — ALGEBRAIC MODELING PROBLEMS
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Algebraic Modeling Problems}}
\end{tcolorbox}

\noindent
Write and evaluate expressions for each word problem.

\vspace{3mm}

\begin{enumerate}[itemsep=1.5cm, leftmargin=0.6cm]

\item A taxi company charges a flat fee of \$3.50 plus \$2.25 per mile driven. \\
\textbf{a)} Write an expression for the cost of a ride of $m$ miles.  \vspace{1.5cm}

\textbf{b)} How much does a ride of 12 miles cost? 

\vspace{1cm}

\item An online store sells shirts for $s$ dollars and pants for $p$ dollars. Shipping is a flat \$7.50. \\

\textbf{a)} Write an expression for the total cost of buying 3 shirts and 2 pairs of pants. \\
\vspace{1.5cm}

\textbf{b)} Find the total cost if $s = 18.00$ and $p = 25.00$. 

\vspace{1.5cm}

\item A rectangular garden has length $2x + 3$ and width $x$. Write an expression for its perimeter. Find the perimeter if $x = 5$ feet.

\vspace{2cm}

\end{enumerate}

\noindent
Explain the errors or solve the conceptual problems below.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item Explain why the expression $12 - 3x$ is evaluated differently for $x = 2$ depending on whether order of operations is followed. What is the correct value?

\vspace{3cm}

\item A student translates the phrase ``half of the sum of a number $y$ and 6'' as $\frac{y}{2} + 6$. Explain the student's error and write the correct expression.

\vspace{3 cm}

\item For what value of the variable $a$ does the expression $5a - 8$ equal $12$? Show your steps.

\vspace{2cm}

\end{enumerate}
