\begin{tcolorbox}[colback=black!10]
    \large\textcolor{black}{\textbf{Mean, Median, Mode, and Range (Foundation)}}
\end{tcolorbox}

\vspace{2mm}

\noindent
1. \textbf{Mean (Average):} The numerical balance point. Sum all values and divide by the total number of values ($\text{Mean} = \frac{\text{Total Sum}}{n}$). 
   \begin{itemize}[noitemsep, leftmargin=0.5cm]
       \item \textit{Finding a Missing Value:} $\text{Required Sum} = \text{Target Mean} \times \text{Total } n$.
   \end{itemize}
2. \textbf{Median:} The absolute center position of an ordered data set.
   \begin{itemize}[noitemsep, leftmargin=0.5cm]
       \item If $n$ is \textbf{odd}: The single middle number.
       \item If $n$ is \textbf{even}: The average of the two central numbers: $\frac{\text{middle}_1 + \text{middle}_2}{2}$.
   \end{itemize}
3. \textbf{Mode \& Range:} The \textbf{Mode} is the value that occurs most frequently. The \textbf{Range} measures overall spread ($\text{Maximum} - \text{Minimum}$).

\vspace{3mm}

\begin{tcolorbox}[colback=white, colframe=black!25, boxrule=0.6pt, arc=4mm, left=6mm, right=6mm, top=4mm, bottom=4mm]
\textbf{Worked Example:}\\
Find the mean and median of the dataset: $3, 4, 4, 5, 9$.\\ \\ \\
\textbf{Solution:}\\ \\
- Mean: $\frac{3 + 4 + 4 + 5 + 9}{5} = \frac{25}{5} = 5.$\\ \\
- Median: The numbers are already ordered: $3, 4, \mathbf{4}, 5, 9$. The center data point is $4$. \\
\end{tcolorbox}
