% ==========================================================
% PART C — EQUATION WORD PROBLEMS
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Equation Word Problems}}
\end{tcolorbox}

\noindent
Write a one-variable equation for each situation and solve it.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item A recipe needs some flour. You already added 2 cups. You still need 3.5 cups. Write an addition equation to find the total flour $F$ needed. Solve it.

\vspace{2cm}

\item You bought 5 identical notebooks and spent \$14.75 in total. Write a multiplication equation to find the cost $c$ of one notebook. Solve it.

\vspace{2cm}

\item A stock price dropped by \$5.20 today, closing at \$42.50. Write a subtraction equation to find the opening price $P$. Solve it.

\vspace{2cm}

\end{enumerate}

\newpage

% ==========================================================
% PART D — GEOMETRIC APPLICATION EQUATIONS
% ==========================================================
\begin{tcolorbox}[colback=black!10]
\large\textcolor{black}{\textbf{Geometric Application Equations}}
\end{tcolorbox}

\noindent
Use geometric concepts and equations to solve.

\vspace{3mm}

\begin{enumerate}[itemsep=2.2cm, leftmargin=0.6cm]

\item The perimeter of an equilateral triangle is 24.6 inches. Write an equation using the side length $s$. Solve it to find $s$.

\vspace{2.2cm}

\item A rectangular pool has a width of 12 meters and an area of 180 square meters. Write a multiplication equation to find the length $l$. Solve it.

\vspace{2.2cm}

\item A rectangular box has a square base with side length 4 cm. Its volume is 80 cubic cm. Write an equation using the height $h$ ($Volume = \text{base area} \times h$) and solve it.

\vspace{2.2cm}

\end{enumerate}
